        function loadQuotations(accountId) {

            var container = document.getElementById("quotations-container");

            var allQuots = [];

            

            // Get parent account name

            ZOHO.CRM.API.getRecord({

                Entity: "Accounts",

                RecordID: accountId

            }).then(function(accountResponse) {

                if (accountResponse.data && accountResponse.data[0]) {

                    parentAccountName = accountResponse.data[0].Account_Name;

                    document.getElementById("accountName").textContent = parentAccountName;

                }

            });

            

            // Get child accounts

            ZOHO.CRM.API.searchRecord({

                Entity: "Accounts",

                Type: "criteria",

                Query: "(Parent_Account:equals:" + accountId + ")"

            }).then(function(childResponse) {

                var childAccountIds = [];

                var accountMap = {};

                

                if (childResponse.data && childResponse.data.length > 0) {

                    childAccountIds = childResponse.data.map(function(child) {

                        accountMap[child.id] = { name: child.Account_Name, isParent: false };

                        return child.id;

                    });

                }

                

                // Mark parent account

                accountMap[accountId] = { name: parentAccountName, isParent: true };

                

                // Add the parent account ID to the list

                var accountIds = [accountId].concat(childAccountIds);

                

                // Get quotations for parent and all child accounts

                var quotationPromises = accountIds.map(function(accId) {

                    return ZOHO.CRM.API.getRelatedRecords({

                        Entity: "Accounts",

                        RecordID: accId,

                        RelatedList: "Quotations",

                        page: 1,

                        per_page: 200

                    }).then(function(response) {

                        if (response.data && response.data.length > 0) {

                            // Tag each quotation with its source account

                            response.data.forEach(function(quot) {

                                quot._sourceAccountId = accId;

                                quot._sourceAccountName = accountMap[accId].name;

                                quot._isFromParent = accountMap[accId].isParent;

                            });

                            return response.data;

                        }

                        return [];

                    }).catch(function() {

                        return [];

                    });

                });

                

                Promise.all(quotationPromises).then(function(quotationArrays) {

                    // Flatten all quotations into one array

                    quotationArrays.forEach(function(quotations) {

                        allQuots = allQuots.concat(quotations);

                    });

                    

                    if (allQuots.length === 0) {

                        container.innerHTML = '<div class="no-quotations">No quotations found for this account or its member accounts</div>';

                        return;

                    }

                    

                    processQuotations(allQuots);

                });

            }).catch(function(error) {

                // If child search fails, just get parent quotations

                ZOHO.CRM.API.getRelatedRecords({

                    Entity: "Accounts",

                    RecordID: accountId,

                    RelatedList: "Quotations",

                    page: 1,

                    per_page: 200

                }).then(function(response) {

                    if (!response.data || response.data.length === 0) {

                        container.innerHTML = '<div class="no-quotations">No quotations found for this account</div>';

                        return;

                    }

                    

                    // Tag parent quotations

                    response.data.forEach(function(quot) {

                        quot._isFromParent = true;

                        quot._sourceAccountName = parentAccountName;

                    });

                    processQuotations(response.data);

                }).catch(function(error) {

                    container.innerHTML = '<div class="error">Error loading quotations. Please try again.</div>';

                });

            });

        }

        

        function processQuotations(quotations) {

            var ebookQuotations = [];

            

            quotations.forEach(function(quot) {

                // Only include accepted quotations that have eBooks

                if (quot.Status === "Accepted" && quot.Includes_eBooks === true) {

                    var expirationDate = quot.eBook_expiration_date;

                    var expirationStatus = getExpirationStatus(expirationDate);

                    var daysRemaining = getDaysRemaining(expirationDate);

                    

                    ebookQuotations.push({

                        quotationId: quot.id,

                        quotationNumber: quot.Subject || quot.Quotation_Number || quot.Quote_Number || "No Number",

                        expirationDate: expirationDate || "No Date Set",

                        expirationStatus: expirationStatus,

                        daysRemaining: daysRemaining,

                        accountName: quot._sourceAccountName || (quot.Account_Name ? quot.Account_Name.name : "Unknown"),

                        isFromParent: quot._isFromParent !== undefined ? quot._isFromParent : true,

                        fulfillmentYear: quot.Fulfillment_Year || "",

                        performanceStart: quot.Performance_Start_Date || "",

                        performanceEnd: quot.Performance_End_Date || "",

                        contactName: quot.Contact_Name ? quot.Contact_Name.name : ""

                    });

                }

            });

            

            allQuotations = ebookQuotations;

            filteredQuotations = ebookQuotations;

            

            if (allQuotations.length === 0) {

                document.getElementById("quotations-container").innerHTML = '<div class="no-quotations">No accepted quotations with eBooks found</div>';

                displayStats();

                return;

            }

            

            displayQuotations();

            createFilters();

            displayStats();

        }

        

        function getExpirationStatus(dateString) {

            if (!dateString) return 'unknown';

            

            var expDate = new Date(dateString);

            var today = new Date();

            var daysUntil = Math.ceil((expDate - today) / (1000 * 60 * 60 * 24));

            

            if (daysUntil < 0) return 'expired';

            if (daysUntil <= 30) return 'expiring';

            return 'active';

        }

        

        function getDaysRemaining(dateString) {

            if (!dateString) return null;

            

            var expDate = new Date(dateString);

            var today = new Date();

            var daysUntil = Math.ceil((expDate - today) / (1000 * 60 * 60 * 24));

            

            return daysUntil;

        }

        

        function formatDate(dateString) {

            if (!dateString) return "No Date Set";

            try {

                return new Date(dateString).toLocaleDateString('en-US', {

                    year: 'numeric',

                    month: 'long',

                    day: 'numeric'

                });

            } catch (error) {

                return dateString;

            }

        }

        

        function displayStats() {

            var total = filteredQuotations.length;

            var active = filteredQuotations.filter(q => q.expirationStatus === 'active').length;

            var expiring = filteredQuotations.filter(q => q.expirationStatus === 'expiring').length;

            var expired = filteredQuotations.filter(q => q.expirationStatus === 'expired').length;

            

            var statsHTML = '<div class="stats-bar">';

            statsHTML += '<div class="stat-item"><div class="stat-value">' + total + '</div><div class="stat-label">Total eBook Quotations</div></div>';

            statsHTML += '<div class="stat-item"><div class="stat-value" style="color: #155724;">' + active + '</div><div class="stat-label">Active</div></div>';

            statsHTML += '<div class="stat-item"><div class="stat-value" style="color: #856404;">' + expiring + '</div><div class="stat-label">Expiring Soon</div></div>';

            statsHTML += '<div class="stat-item"><div class="stat-value" style="color: #721c24;">' + expired + '</div><div class="stat-label">Expired</div></div>';

            statsHTML += '</div>';

            

            document.getElementById("stats-container").innerHTML = statsHTML;

        }

        

        function createFilters() {

            var accounts = [...new Set(allQuotations.map(q => q.accountName))].sort();

            var years = [...new Set(allQuotations.map(q => q.fulfillmentYear).filter(y => y))].sort().reverse();

            

            var filtersHTML = '<div class="filters">';

            

            filtersHTML += '<div class="filter-group">';

            filtersHTML += '<label>Status</label>';

            filtersHTML += '<select id="statusFilter" onchange="applyFilters()">';

            filtersHTML += '<option value="">All Statuses</option>';

            filtersHTML += '<option value="active">Active</option>';

            filtersHTML += '<option value="expiring">Expiring Soon (30 days)</option>';

            filtersHTML += '<option value="expired">Expired</option>';

            filtersHTML += '</select></div>';

            

            filtersHTML += '<div class="filter-group">';

            filtersHTML += '<label>Account</label>';

            filtersHTML += '<select id="accountFilter" onchange="applyFilters()">';

            filtersHTML += '<option value="">All Accounts</option>';

            accounts.forEach(function(name) {

                filtersHTML += '<option value="' + name + '">' + name + '</option>';

            });

            filtersHTML += '</select></div>';

            

            if (years.length > 0) {

                filtersHTML += '<div class="filter-group">';

                filtersHTML += '<label>Fulfillment Year</label>';

                filtersHTML += '<select id="yearFilter" onchange="applyFilters()">';

                filtersHTML += '<option value="">All Years</option>';

                years.forEach(function(year) {

                    filtersHTML += '<option value="' + year + '">' + year + '</option>';

                });

                filtersHTML += '</select></div>';

            }

            

            filtersHTML += '<div class="filter-group">';

            filtersHTML += '<label>Search</label>';

            filtersHTML += '<input type="text" id="searchFilter" placeholder="Quotation number..." onkeyup="applyFilters()">';

            filtersHTML += '</div>';

            

            filtersHTML += '</div>';

            

            document.getElementById("filters-container").innerHTML = filtersHTML;

        }

        

        function applyFilters() {

            var statusFilter = document.getElementById("statusFilter") ? document.getElementById("statusFilter").value : "";

            var accountFilter = document.getElementById("accountFilter") ? document.getElementById("accountFilter").value : "";

            var yearFilter = document.getElementById("yearFilter") ? document.getElementById("yearFilter").value : "";

            var searchFilter = document.getElementById("searchFilter") ? document.getElementById("searchFilter").value.toLowerCase() : "";

            

            filteredQuotations = allQuotations.filter(function(quot) {

                var matchStatus = !statusFilter || quot.expirationStatus === statusFilter;

                var matchAccount = !accountFilter || quot.accountName === accountFilter;

                var matchYear = !yearFilter || quot.fulfillmentYear === yearFilter;

                var matchSearch = !searchFilter || quot.quotationNumber.toLowerCase().includes(searchFilter);

                

                return matchStatus && matchAccount && matchYear && matchSearch;

            });

            

            displayQuotations();

            displayStats();

        }

        

        function displayQuotations() {

            var container = document.getElementById("quotations-container");

            

            if (filteredQuotations.length === 0) {

                container.innerHTML = '<div class="no-quotations">No quotations match your filters</div>';

                return;

            }

            

            var activeQuots = filteredQuotations.filter(q => q.expirationStatus === 'active');

            var expiringQuots = filteredQuotations.filter(q => q.expirationStatus === 'expiring');

            var expiredQuots = filteredQuotations.filter(q => q.expirationStatus === 'expired');

            

            var html = '';

            

            if (activeQuots.length > 0) {

                html += '<div class="status-section">';

                html += '<div class="status-header active">';

                html += '<span>✅ Active eBooks</span>';

                html += '<span class="status-count">' + activeQuots.length + ' quotation' + (activeQuots.length !== 1 ? 's' : '') + '</span>';

                html += '</div>';

                html += renderQuotations(activeQuots);

                html += '</div>';

            }

            

            if (expiringQuots.length > 0) {

                html += '<div class="status-section">';

                html += '<div class="status-header expiring">';

                html += '<span>⚠️ Expiring Soon (within 30 days)</span>';

                html += '<span class="status-count">' + expiringQuots.length + ' quotation' + (expiringQuots.length !== 1 ? 's' : '') + '</span>';

                html += '</div>';

                html += renderQuotations(expiringQuots);

                html += '</div>';

            }

            

            if (expiredQuots.length > 0) {

                html += '<div class="status-section">';

                html += '<div class="status-header expired">';

                html += '<span>❌ Expired eBooks</span>';

                html += '<span class="status-count">' + expiredQuots.length + ' quotation' + (expiredQuots.length !== 1 ? 's' : '') + '</span>';

                html += '</div>';

                html += renderQuotations(expiredQuots);

                html += '</div>';

            }

            

            container.innerHTML = html;

        }

        

        function renderQuotations(quotations) {

            var html = '';

            

            quotations.forEach(function(quot) {

                html += '<div class="quotation-card">';

                html += '<div class="quotation-header">';

                html += '<div class="quotation-number" onclick="openQuotation(\'' + quot.quotationId + '\')">' + quot.quotationNumber + '</div>';

                var badgeClass = quot.isFromParent ? 'account-badge' : 'account-badge member';

                var badgeText = quot.isFromParent ? 'PARENT' : 'MEMBER';

                html += '<div class="' + badgeClass + '">' + badgeText + '</div>';

                html += '</div>';

                

                html += '<div class="quotation-details">';

                

                html += '<div class="detail-item">';

                html += '<div class="detail-label">📚 eBook Expiration Date</div>';

                html += '<div class="detail-value expiration-date ' + quot.expirationStatus + '">' + formatDate(quot.expirationDate) + '</div>';

                if (quot.daysRemaining !== null) {

                    var daysText = quot.daysRemaining >= 0 

                        ? quot.daysRemaining + ' days remaining'

                        : Math.abs(quot.daysRemaining) + ' days overdue';

                    html += '<div class="days-remaining">' + daysText + '</div>';

                }

                html += '</div>';

                

                html += '<div class="detail-item">';

                html += '<div class="detail-label">🏢 Account</div>';

                html += '<div class="detail-value">' + quot.accountName + '</div>';

                html += '</div>';

                

                if (quot.fulfillmentYear) {

                    html += '<div class="detail-item">';

                    html += '<div class="detail-label">📅 Fulfillment Year</div>';

                    html += '<div class="detail-value">' + quot.fulfillmentYear + '</div>';

                    html += '</div>';

                }

                

                if (quot.performanceStart && quot.performanceEnd) {

                    html += '<div class="detail-item">';

                    html += '<div class="detail-label">⏱️ Performance Period</div>';

                    html += '<div class="detail-value">' + formatDate(quot.performanceStart) + ' - ' + formatDate(quot.performanceEnd) + '</div>';

                    html += '</div>';

                }

                

                if (quot.contactName) {

                    html += '<div class="detail-item">';

                    html += '<div class="detail-label">👤 Contact</div>';

                    html += '<div class="detail-value">' + quot.contactName + '</div>';

                    html += '</div>';

                }

                

                html += '</div>';

                html += '</div>';

            });

            

            return html;

        }
